Subqueries

ALL and ANY

-- earn more than EVERY employee in dept 5

SELECT lname, fname

FROM   employee

WHERE  salary > ALL (SELECT salary

                     FROM   employee

                     WHERE  dno = 5);

 

-- the same with an aggregate:

WHERE  salary > (SELECT MAX(salary)

                 FROM employee WHERE dno = 5);

FormMeans
> ALL (…)> the maximum
< ALL (…)< the minimum
> ANY (…)> the minimum
= ANY (…)same as IN
