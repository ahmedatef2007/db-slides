Aggregation

HAVING filters groups

-- projects with more than two employees

SELECT   p.pnumber, p.pname, COUNT(*) AS staff

FROM     project  p

JOIN     works_on w ON w.pno = p.pnumber

GROUP BY p.pnumber, p.pname

HAVING   COUNT(*) > 2

ORDER BY p.pnumber;
