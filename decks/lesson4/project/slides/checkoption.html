Views

WITH CHECK OPTION

CREATE VIEW v_supplier AS

SELECT *

FROM   suppliers

WHERE  status > 15

WITH CHECK OPTION;

 

-- rejected: the new row would disappear from the view

UPDATE v_supplier SET status = 10 WHERE s_id = 'S1';

INSERTs and UPDATEs through the view must produce rows that still satisfy the view's WHERE (status > 15).
