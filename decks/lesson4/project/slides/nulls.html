NULL: the missing value

  • NULL means unknown or not applicable, not 0 or ''
  • Any comparison with NULL is UNKNOWN, so the row is filtered out
  • Use IS NULL / IS NOT NULL, never = NULL
  • COALESCE(x, default) replaces NULL with a default

-- employees with no supervisor

SELECT fname FROM employee

WHERE  super_ssn IS NULL;

 

-- WRONG: returns nothing

SELECT fname FROM employee

WHERE  super_ssn = NULL;

 

SELECT fname, COALESCE(super_ssn, 'none')

FROM   employee;
